2.Even Number and Range Checker

import java.util.*;
import java.util.Scanner;
public class evennumberandrangechecker
{
    public static void main(String[] args)
    {
        System.out.println("Enter a number: ");
        Scanner scanner = new Scanner(System.in);
        int number = scanner.nextInt();
        if (number % 2 == 0 && number >= 10)
        {
            System.out.println(number + " is an even number and greater than 10.");
        }
        else if (number % 2 == 0 && number <= 10)
        {
            System.out.println(number + " is  an even number and not greater than 10.");
        }
        else
        {
            System.out.println(number + " is not an even number and not greater than 10.");
        }
    }
}

12.Ride Fare Calculator


import java.util.Scanner;

public class rideFarecalculator {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        double distance, fare;
        int waiting, extraTime;

        System.out.print("Enter distance in km: ");
        distance = sc.nextDouble();

        System.out.print("Enter waiting time in minutes: ");
        waiting = sc.nextInt();
        sc.nextLine(); // Consume newline

        System.out.print("Premium member? (Yes/No): ");
        String premiumInput = sc.nextLine();

        if (distance <= 0 || waiting < 0) {
            System.out.println("Invalid input!");
        } else {
            extraTime = (waiting > 5) ? waiting - 5 : 0;

            fare = 50;
            fare += distance * 12;
            fare += extraTime * 2;

            fare = (premiumInput.equalsIgnoreCase("yes")
                    && fare > 300) ? fare * 0.90 : fare;

            System.out.printf("Total Ride Fare: Rs. %.2f%n", fare);
        }

        sc.close();
    }
}



13.Employee Bonus Calculator


import java.util.Scanner;

public class employeebonuscalculator {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        double salary, bonus, finalSalary;
        int score, projects, bonusPercent;

        System.out.print("Enter employee salary: ");
        salary = sc.nextDouble();

        System.out.print("Enter performance score: ");
        score = sc.nextInt();

        System.out.print("Enter completed projects: ");
        projects = sc.nextInt();

        bonusPercent = (score >= 90 && projects >= 3) ? 15 :
                       (score >= 75 || projects >= 5) ? 10 :
                       (score >= 60) ? 3 : 0;

        bonus = salary * bonusPercent / 100;
        finalSalary = salary + bonus;

        System.out.println("Bonus Percentage: " + bonusPercent + "%");
        System.out.printf("Bonus Amount: Rs. %.2f%n", bonus);
        System.out.printf("Final Salary: Rs. %.2f%n", finalSalary);

        sc.close();
    }
}


14.Secure Transaction Validator


import java.util.Scanner;

public class securetransactionvalidator {
    public static void main(String[] args) {
        Scanner sc = new Scanner(System.in);

        double balance, amount, dailyLimit;
        boolean active;

        System.out.print("Enter account balance: ");
        balance = sc.nextDouble();

        System.out.print("Enter transfer amount: ");
        amount = sc.nextDouble();

        System.out.print("Enter daily limit: ");
        dailyLimit = sc.nextDouble();

        System.out.print("Is account active? (true/false): ");
        active = sc.nextBoolean();

        if (amount > 0 && amount <= balance &&
            amount <= dailyLimit && active) {

            balance -= amount;
            System.out.println("Transaction Successful!");
        } else {
            System.out.println("Transaction Failed!");
        }

        System.out.printf("Remaining Balance: Rs. %.2f%n", balance);

        sc.close();
    }
}
